# Which struct member each memory access of the toy's demo_* functions touches, in btf2ida's layout.
# Written by tests/check_toy_members.py from `objdump -d -M intel` of demo/toy_kernel.elf and the toy's BTF;
# derived offline, not IDA output. After the import the decompiler should show these names instead of offsets.
# Bit numbers count from the accessed address. The flexible array `payload` is only a comment in the header,
# so its loop in demo_checksum stays a raw offset.
#
# function  address  instruction  ->  member
demo_open      0x112d  mov DWORD PTR [rdi+0xc],0x5          ->  dev->state
demo_open      0x113a  movzx esi,BYTE PTR [rdi+0x8]         ->  dev->ready (bit 0), dev->mode (bits 1-3), dev->refs (bits 4-15); then `and 0xfffffff1` clears mode
demo_open      0x1143  mov BYTE PTR [rdi+0x8],sil           ->  dev->mode (the field the mask cleared)
demo_open      0x1147  movzx edx,WORD PTR [rdi+0x8]         ->  dev->ready (bit 0), dev->mode (bits 1-3), dev->refs (bits 4-15); then `shr 0x4` reads refs; then `and 0xf` clears refs
demo_open      0x115c  mov WORD PTR [rdi+0x8],dx            ->  dev->refs (the field the mask cleared)
demo_close     0x1168  mov DWORD PTR [rdi+0xc],0x6          ->  dev->state
